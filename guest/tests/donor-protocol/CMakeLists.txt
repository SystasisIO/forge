cmake_minimum_required(VERSION 3.31)
project(ForgeDonorProtocolTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT TARGET forge_chain_protocol)
   find_package(Forge CONFIG REQUIRED COMPONENTS chain_protocol)
   set(_protocol_owner Forge::forge_chain_protocol)
else()
   set(_protocol_owner forge_chain_protocol)
endif()
find_package(Boost REQUIRED)
add_executable(forge_contract_protocol_value_tests protocol_value_tests.cpp)
target_link_libraries(forge_contract_protocol_value_tests PRIVATE ${_protocol_owner} Boost::headers)
enable_testing()
add_test(NAME forge_contract_protocol_value_tests COMMAND forge_contract_protocol_value_tests)

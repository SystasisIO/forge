cmake_minimum_required(VERSION 3.31)

project(ForgeContractE2ETests LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT FORGE_CONTRACT_TEST_WASM OR NOT EXISTS "${FORGE_CONTRACT_TEST_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_WASM must name a built contract")
endif()
if(NOT FORGE_CONTRACT_TEST_RECORD_WASM OR NOT EXISTS "${FORGE_CONTRACT_TEST_RECORD_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_RECORD_WASM must name the record contract")
endif()
if(NOT FORGE_CONTRACT_TEST_GUEST_MACRO_WASM OR NOT EXISTS "${FORGE_CONTRACT_TEST_GUEST_MACRO_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_GUEST_MACRO_WASM must name the declaration-definition contract")
endif()
if(NOT FORGE_CONTRACT_TEST_ALLOCATOR_WASM OR NOT EXISTS "${FORGE_CONTRACT_TEST_ALLOCATOR_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_ALLOCATOR_WASM must name the allocator contract")
endif()
if(NOT FORGE_CONTRACT_TEST_LEGACY_WASM OR NOT EXISTS "${FORGE_CONTRACT_TEST_LEGACY_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_LEGACY_WASM must name the legacy contract")
endif()
if(NOT FORGE_CONTRACT_TEST_LEGACY_NOTIFICATION_WASM OR NOT EXISTS "${FORGE_CONTRACT_TEST_LEGACY_NOTIFICATION_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_LEGACY_NOTIFICATION_WASM must name the legacy notification contract")
endif()
if(NOT FORGE_CONTRACT_TEST_DB_WASM OR NOT EXISTS "${FORGE_CONTRACT_TEST_DB_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_DB_WASM must name the database intrinsic contract")
endif()
if(NOT FORGE_CONTRACT_TEST_DB_HOST_WASM OR NOT EXISTS "${FORGE_CONTRACT_TEST_DB_HOST_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_DB_HOST_WASM must name the executable database host contract")
endif()
if(NOT FORGE_CONTRACT_TEST_ORACLE_WASM OR NOT EXISTS "${FORGE_CONTRACT_TEST_ORACLE_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_ORACLE_WASM must name the complete intrinsic oracle contract")
endif()
if(NOT FORGE_CONTRACT_TEST_RECOVERY_WASM OR NOT EXISTS "${FORGE_CONTRACT_TEST_RECOVERY_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_RECOVERY_WASM must name the public-key recovery contract")
endif()
if(NOT FORGE_CONTRACT_TEST_LOOPING_WASM OR NOT EXISTS "${FORGE_CONTRACT_TEST_LOOPING_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_LOOPING_WASM must name the bounded-execution contract")
endif()
if(NOT FORGE_CONTRACT_TEST_MULTI_INDEX_WASM OR NOT EXISTS "${FORGE_CONTRACT_TEST_MULTI_INDEX_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_MULTI_INDEX_WASM must name the multi_index contract")
endif()
if(NOT FORGE_CONTRACT_TEST_LEGACY_MULTI_INDEX_WASM OR NOT EXISTS "${FORGE_CONTRACT_TEST_LEGACY_MULTI_INDEX_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_LEGACY_MULTI_INDEX_WASM must name the legacy multi_index contract")
endif()
if(NOT FORGE_CONTRACT_TEST_MULTI_INDEX_ERRORS_WASM OR NOT EXISTS "${FORGE_CONTRACT_TEST_MULTI_INDEX_ERRORS_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_MULTI_INDEX_ERRORS_WASM must name the multi_index error contract")
endif()
if(NOT FORGE_CONTRACT_TEST_LEGACY_MULTI_INDEX_ERRORS_WASM OR
   NOT EXISTS "${FORGE_CONTRACT_TEST_LEGACY_MULTI_INDEX_ERRORS_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_LEGACY_MULTI_INDEX_ERRORS_WASM must name the legacy error contract")
endif()
if(NOT FORGE_CONTRACT_TEST_MULTI_INDEX_EXTENDED_WASM OR
   NOT EXISTS "${FORGE_CONTRACT_TEST_MULTI_INDEX_EXTENDED_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_MULTI_INDEX_EXTENDED_WASM must name the extended multi_index contract")
endif()
if(NOT FORGE_CONTRACT_TEST_LEGACY_MULTI_INDEX_EXTENDED_WASM OR
   NOT EXISTS "${FORGE_CONTRACT_TEST_LEGACY_MULTI_INDEX_EXTENDED_WASM}")
   message(FATAL_ERROR "FORGE_CONTRACT_TEST_LEGACY_MULTI_INDEX_EXTENDED_WASM must name the legacy extended contract")
endif()

find_package(
   Forge CONFIG REQUIRED
   COMPONENTS chain_protocol tooling_testing crypto_asymmetric crypto_digest raw vm_wasm_interpret
)
find_package(Boost REQUIRED)

add_executable(forge_contract_e2e_tests contract_tests.cpp)
target_link_libraries(
   forge_contract_e2e_tests
   PRIVATE
      Forge::forge_chain_protocol
      Forge::forge_crypto_asymmetric
      Forge::forge_crypto_digest
      Forge::forge_tooling_testing
      Forge::forge_raw
      Forge::forge_vm_wasm_interpret
      Boost::headers
)
target_compile_definitions(
   forge_contract_e2e_tests
   PRIVATE
      FORGE_CONTRACT_TEST_WASM="${FORGE_CONTRACT_TEST_WASM}"
      FORGE_CONTRACT_TEST_RECORD_WASM="${FORGE_CONTRACT_TEST_RECORD_WASM}"
      FORGE_CONTRACT_TEST_GUEST_MACRO_WASM="${FORGE_CONTRACT_TEST_GUEST_MACRO_WASM}"
      FORGE_CONTRACT_TEST_ALLOCATOR_WASM="${FORGE_CONTRACT_TEST_ALLOCATOR_WASM}"
      FORGE_CONTRACT_TEST_LEGACY_WASM="${FORGE_CONTRACT_TEST_LEGACY_WASM}"
      FORGE_CONTRACT_TEST_LEGACY_NOTIFICATION_WASM="${FORGE_CONTRACT_TEST_LEGACY_NOTIFICATION_WASM}"
      FORGE_CONTRACT_TEST_DB_WASM="${FORGE_CONTRACT_TEST_DB_WASM}"
      FORGE_CONTRACT_TEST_DB_HOST_WASM="${FORGE_CONTRACT_TEST_DB_HOST_WASM}"
      FORGE_CONTRACT_TEST_ORACLE_WASM="${FORGE_CONTRACT_TEST_ORACLE_WASM}"
      FORGE_CONTRACT_TEST_RECOVERY_WASM="${FORGE_CONTRACT_TEST_RECOVERY_WASM}"
      FORGE_CONTRACT_TEST_LOOPING_WASM="${FORGE_CONTRACT_TEST_LOOPING_WASM}"
      FORGE_CONTRACT_TEST_MULTI_INDEX_WASM="${FORGE_CONTRACT_TEST_MULTI_INDEX_WASM}"
      FORGE_CONTRACT_TEST_LEGACY_MULTI_INDEX_WASM="${FORGE_CONTRACT_TEST_LEGACY_MULTI_INDEX_WASM}"
      FORGE_CONTRACT_TEST_MULTI_INDEX_ERRORS_WASM="${FORGE_CONTRACT_TEST_MULTI_INDEX_ERRORS_WASM}"
      FORGE_CONTRACT_TEST_LEGACY_MULTI_INDEX_ERRORS_WASM="${FORGE_CONTRACT_TEST_LEGACY_MULTI_INDEX_ERRORS_WASM}"
      FORGE_CONTRACT_TEST_MULTI_INDEX_EXTENDED_WASM="${FORGE_CONTRACT_TEST_MULTI_INDEX_EXTENDED_WASM}"
      FORGE_CONTRACT_TEST_LEGACY_MULTI_INDEX_EXTENDED_WASM="${FORGE_CONTRACT_TEST_LEGACY_MULTI_INDEX_EXTENDED_WASM}"
      FORGE_CONTRACT_TEST_DB_GOLDEN="${CMAKE_CURRENT_SOURCE_DIR}/../tooling/spring_db_intrinsics.txt"
)

enable_testing()
add_test(NAME forge_contract_e2e_tests COMMAND forge_contract_e2e_tests)

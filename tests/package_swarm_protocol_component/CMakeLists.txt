cmake_minimum_required(VERSION 3.31)
project(forge-package-swarm-protocol-component LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_DISABLE_FIND_PACKAGE_Ngtcp2 ON)
set(CMAKE_DISABLE_FIND_PACKAGE_Notcurses ON)
set(CMAKE_DISABLE_FIND_PACKAGE_RocksDB ON)
find_package(Forge CONFIG REQUIRED COMPONENTS swarm_protocol)
include(${CMAKE_CURRENT_LIST_DIR}/../package_dependency_boundary.cmake)
forge_package_assert_dependency_boundary(Forge::forge_swarm_protocol
   "Forge::forge_(swarm_(db|storage|runtime|api|conversion)|db|net|api|asio|app|plugins|chain)($|_)")
add_executable(forge_package_swarm_protocol_component main.cpp)
target_link_libraries(forge_package_swarm_protocol_component PRIVATE Forge::forge_swarm_protocol)

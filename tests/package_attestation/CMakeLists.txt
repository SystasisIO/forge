cmake_minimum_required(VERSION 3.31)
project(ForgePackageAttestation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Forge CONFIG REQUIRED COMPONENTS core)
if(TARGET cjose::cjose_static)
   message(FATAL_ERROR "Core-only lookup must not discover the JOSE backend")
endif()
find_package(Forge CONFIG REQUIRED COMPONENTS auth_attestation auth_api crypto_jose codec_json)
get_target_property(_attestation_links_before Forge::forge_crypto_jose INTERFACE_LINK_LIBRARIES)
if(NOT "$<LINK_ONLY:cjose::cjose_static>" IN_LIST _attestation_links_before
   OR "cjose::cjose_static" IN_LIST _attestation_links_before)
   message(FATAL_ERROR "JOSE must retain its private qualified backend link")
endif()
find_package(Forge CONFIG REQUIRED COMPONENTS crypto_jose)
get_target_property(_attestation_links_after Forge::forge_crypto_jose INTERFACE_LINK_LIBRARIES)
if(NOT _attestation_links_before STREQUAL _attestation_links_after)
   message(FATAL_ERROR "Repeated JOSE lookup changed the qualified backend binding")
endif()

# Override expectations only, never discovery, for a separate external-backend consumer.
set(_attestation_cjose_libdir "${Forge_DIR}/../..")
if(DEFINED FORGE_ATTESTATION_EXPECTED_CJOSE_LIBDIR)
   set(_attestation_cjose_libdir "${FORGE_ATTESTATION_EXPECTED_CJOSE_LIBDIR}")
endif()
file(REAL_PATH "${_attestation_cjose_libdir}" _attestation_cjose_libdir)
file(REAL_PATH "${cjose_DIR}" _attestation_cjose_dir)
if(NOT _attestation_cjose_dir STREQUAL "${_attestation_cjose_libdir}/cmake/cjose")
   message(FATAL_ERROR "CJOSE package resolved outside the expected installed libdir: ${cjose_DIR}")
endif()
if(NOT TARGET cjose::cjose_static)
   message(FATAL_ERROR "CJOSE package did not provide its static target")
endif()
get_target_property(_attestation_cjose_imported cjose::cjose_static IMPORTED)
get_target_property(_attestation_cjose_type cjose::cjose_static TYPE)
if(NOT _attestation_cjose_imported OR NOT _attestation_cjose_type STREQUAL "STATIC_LIBRARY")
   message(FATAL_ERROR "CJOSE static target must come from an installed package, not the build tree")
endif()

get_target_property(_attestation_cjose_configs cjose::cjose_static IMPORTED_CONFIGURATIONS)
set(_attestation_cjose_locations IMPORTED_LOCATION)
if(_attestation_cjose_configs)
   foreach(config IN LISTS _attestation_cjose_configs)
      string(TOUPPER "${config}" config)
      list(APPEND _attestation_cjose_locations "IMPORTED_LOCATION_${config}")
   endforeach()
endif()
set(_attestation_cjose_has_archive FALSE)
foreach(location IN LISTS _attestation_cjose_locations)
   get_target_property(_attestation_cjose_archive cjose::cjose_static "${location}")
   if(_attestation_cjose_archive)
      file(REAL_PATH "${_attestation_cjose_archive}" _attestation_cjose_archive)
      cmake_path(IS_PREFIX _attestation_cjose_libdir "${_attestation_cjose_archive}"
         NORMALIZE _attestation_cjose_owned)
      if(NOT _attestation_cjose_owned OR NOT EXISTS "${_attestation_cjose_archive}"
         OR IS_DIRECTORY "${_attestation_cjose_archive}")
         message(FATAL_ERROR "CJOSE ${location} is not an archive under the expected installed libdir: ${_attestation_cjose_archive}")
      endif()
      set(_attestation_cjose_has_archive TRUE)
   endif()
endforeach()
if(NOT _attestation_cjose_has_archive)
   message(FATAL_ERROR "CJOSE static target has no imported archive location")
endif()

foreach(component forge_auth_attestation forge_auth_api forge_crypto_jose)
   get_target_property(links Forge::${component} INTERFACE_LINK_LIBRARIES)
   if(links MATCHES "forge_(chain|net|plugins)_|oauth2-internal|CURL::|PCRE2")
      message(FATAL_ERROR "${component} exports an unrelated runtime/backend dependency")
   endif()
endforeach()
if(TARGET OAuth2::OAuth2 OR TARGET oauth2-internal OR TARGET CURL::libcurl)
   message(FATAL_ERROR "The attestation-only package consumer discovered OAuth transport dependencies")
endif()

add_executable(forge_package_attestation main.cpp)
target_link_libraries(forge_package_attestation PRIVATE
   Forge::forge_auth_api Forge::forge_crypto_jose Forge::forge_codec_json)
enable_testing()
add_test(NAME test_forge_package_attestation COMMAND forge_package_attestation)

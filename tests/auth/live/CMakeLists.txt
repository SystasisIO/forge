cmake_minimum_required(VERSION 3.31)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
   project(forge-auth-live LANGUAGES CXX)
endif()

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Opt-in executable only: building never starts a browser or performs a login.
if(APPLE)
   enable_language(OBJCXX)
   if(TARGET forge_auth_appauth)
      set(_forge_live_backend forge_auth_appauth)
   else()
      find_package(Forge CONFIG REQUIRED COMPONENTS auth_appauth)
      set(_forge_live_backend Forge::forge_auth_appauth)
   endif()
   find_library(LIVE_APPKIT AppKit REQUIRED)
   add_executable(forge_auth_live_appauth macos_driver.mm appauth_driver.cpp common.cpp)
   set_source_files_properties(macos_driver.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
   set_target_properties(forge_auth_live_appauth PROPERTIES
      OBJCXX_STANDARD 23 OBJCXX_STANDARD_REQUIRED YES MACOSX_DEPLOYMENT_TARGET "12.0")
   target_link_libraries(forge_auth_live_appauth PRIVATE ${_forge_live_backend} ${LIVE_APPKIT})
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
   if(TARGET forge_auth_oidc_agent)
      set(_forge_live_backend forge_auth_oidc_agent)
   else()
      find_package(Forge CONFIG REQUIRED COMPONENTS auth_oidc_agent)
      set(_forge_live_backend Forge::forge_auth_oidc_agent)
   endif()
   foreach(field OIDC_AGENT_LINK_LIBRARIES OIDC_AGENT_LDFLAGS OIDC_AGENT_LIBRARY_DIRS)
      if("${${field}}" MATCHES "forge_test|package_auth_oidc_agent|oidc_agent_stub")
         message(FATAL_ERROR "Package-test oidc-agent stubs are not live providers")
      endif()
   endforeach()
   add_executable(forge_auth_live_oidc_agent linux_driver.cpp common.cpp)
   target_link_libraries(forge_auth_live_oidc_agent PRIVATE ${_forge_live_backend})
else()
   message(FATAL_ERROR "Live fixtures support native macOS and Linux only")
endif()

find_package(Threads REQUIRED)
if(APPLE)
   target_link_libraries(forge_auth_live_appauth PRIVATE Threads::Threads)
else()
   target_link_libraries(forge_auth_live_oidc_agent PRIVATE Threads::Threads)
endif()

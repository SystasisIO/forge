cmake_minimum_required(VERSION 3.22)
project(cjose_forge_feature_detection LANGUAGES C)
option(EXPECT_EXTENSION "Require candidate extension, or prove its absence in upstream" ON)
option(EXPECT_JANSSON_FALLBACK "Require pkg-config/manual Jansson discovery instead of native config" OFF)
list(PREPEND CMAKE_MODULE_PATH "${FORGE_CMAKE_MODULES}")
find_package(CJOSE QUIET MODULE)
if(NOT cjose_FOUND OR NOT FORGE_CJOSE_HAS_QUALIFIED_DEPENDENCIES)
   message(FATAL_ERROR "Native package and qualified dependency control are required in both cases")
endif()
if(NOT DEFINED FORGE_CJOSE_HAS_EXTERNAL_SIGNING)
   message(FATAL_ERROR "The extension probe must actually execute after dependency control")
endif()
if(EXPECT_EXTENSION)
   if(NOT CJOSE_FOUND OR NOT FORGE_CJOSE_HAS_EXTERNAL_SIGNING)
      message(FATAL_ERROR "Candidate must provide the qualified extension")
   endif()
   # Only the patched package is required to support repeated native lookup.
   # Never alter or repeatedly load the unmodified upstream package.
   find_package(cjose 0.8.1 EXACT CONFIG REQUIRED COMPONENTS static)
   find_package(cjose 0.8.1 EXACT CONFIG REQUIRED COMPONENTS static)
   # COMPONENTS are required by default; QUIET must report false rather than
   # abort configuration for an unknown required component.
   find_package(cjose 0.8.1 EXACT QUIET CONFIG COMPONENTS qualification_unknown_required)
   if(cjose_FOUND)
      message(FATAL_ERROR "Unknown required native component must be rejected")
   endif()
   find_package(cjose 0.8.1 EXACT CONFIG REQUIRED COMPONENTS static)
   if(NOT cjose_FOUND OR NOT TARGET cjose::cjose_static)
      message(FATAL_ERROR "Valid native component discovery must recover")
   endif()
   list(PREPEND CMAKE_MODULE_PATH "${cjose_DIR}")
   set(saved_include_path "${CMAKE_INCLUDE_PATH}")
   if(NOT EXPECT_JANSSON_FALLBACK)
      list(PREPEND CMAKE_INCLUDE_PATH "${CMAKE_CURRENT_LIST_DIR}/alternate_include")
   endif()
   find_package(Jansson 2.14.0 EXACT QUIET MODULE)
   if(Jansson_FOUND)
      message(FATAL_ERROR "Mismatched exact Jansson release must be rejected")
   endif()
   find_package(Jansson 2.14.1 EXACT REQUIRED MODULE)
   set(CMAKE_INCLUDE_PATH "${saved_include_path}")
   if(NOT Jansson_FOUND OR NOT Jansson_VERSION VERSION_EQUAL "2.14.1")
      message(FATAL_ERROR "Selected Jansson must report its exact header release version")
   endif()
   if(EXPECT_JANSSON_FALLBACK)
      if(NOT CMAKE_DISABLE_FIND_PACKAGE_jansson OR TARGET jansson::jansson)
         message(FATAL_ERROR "Fallback test must not load the native Jansson config")
      endif()
      file(REAL_PATH "${QUALIFICATION_DEPENDENCY_PREFIX}/include" expected_include)
      file(REAL_PATH "${Jansson_INCLUDE_DIR}" actual_include)
      file(REAL_PATH "${QUALIFICATION_DEPENDENCY_PREFIX}/lib" expected_lib)
      file(REAL_PATH "${Jansson_LIBRARY}" actual_library)
      get_filename_component(actual_lib "${actual_library}" DIRECTORY)
      if(NOT actual_include STREQUAL expected_include OR NOT actual_lib STREQUAL expected_lib)
         message(FATAL_ERROR "Fallback discovery must select the same pinned dependency prefix")
      endif()
   elseif(NOT TARGET jansson::jansson)
      message(FATAL_ERROR "Primary candidate test requires native Jansson config discovery")
   endif()
else()
   if(CJOSE_FOUND OR FORGE_CJOSE_HAS_EXTERNAL_SIGNING)
      message(FATAL_ERROR "Unmodified upstream must lack the qualified extension")
   endif()
endif()
message(STATUS "QUALIFICATION_FEATURE_RESULT_PASS")

cmake_minimum_required(VERSION 3.31)
project(forge-package-swarm-api-component LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_DISABLE_FIND_PACKAGE_Ngtcp2 ON)
set(CMAKE_DISABLE_FIND_PACKAGE_Notcurses ON)
set(CMAKE_DISABLE_FIND_PACKAGE_RocksDB ON)
find_package(Forge CONFIG REQUIRED COMPONENTS swarm_api)
include(${CMAKE_CURRENT_LIST_DIR}/../package_dependency_boundary.cmake)
forge_package_assert_dependency_boundary(Forge::forge_swarm_api
   "Forge::forge_(swarm_(db|storage|runtime|conversion)|db|net|app|plugins|chain|api_(p2p|http|quic|websocket|transport|stream))($|_)")
add_executable(forge_package_swarm_api_component main.cpp)
target_link_libraries(forge_package_swarm_api_component PRIVATE Forge::forge_swarm_api)

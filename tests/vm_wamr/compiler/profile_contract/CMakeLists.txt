cmake_minimum_required(VERSION 3.31)
project(forge_wamr_profile_contract LANGUAGES NONE)
include("${PROFILE}")
foreach(_setting IN ITEMS MEMORY_PROFILING THREAD_MGR ALLOC_WITH_USAGE)
   if(NOT WAMR_BUILD_${_setting} STREQUAL "1")
      message(FATAL_ERROR "Required usage-aware runtime feature disabled: ${_setting}")
   endif()
endforeach()
foreach(_setting IN ITEMS GC LOAD_CUSTOM_SECTION QUICK_AOT_ENTRY SHRUNK_MEMORY)
   if(NOT WAMR_BUILD_${_setting} STREQUAL "0")
      message(FATAL_ERROR "Unsupported runtime feature enabled: ${_setting}")
   endif()
endforeach()

set_property(DIRECTORY PROPERTY COMPILE_DEFINITIONS
   WASM_ENABLE_GC=1 WASM_MEM_ALLOC_WITH_USAGE=0 WASM_ENABLE_MEMORY_PROFILING=0
   WASM_ENABLE_MODULE_INST_CONTEXT=1 WASM_ENABLE_WAMR_COMPILER=99 UNRELATED_FLAG=7)
forge_wamr_apply_profile("${CMAKE_CURRENT_SOURCE_DIR}" runtime)
get_directory_property(_runtime COMPILE_DEFINITIONS)
foreach(_required IN ITEMS WASM_ENABLE_GC=0 WASM_MEM_ALLOC_WITH_USAGE=1
      WASM_MEM_ALLOC_WITH_USER_DATA=0 WASM_ENABLE_MEMORY_PROFILING=1
      WASM_ENABLE_THREAD_MGR=1 WASM_ENABLE_MODULE_INST_CONTEXT=0
      WASM_ENABLE_LOAD_CUSTOM_SECTION=0 WASM_ENABLE_WAMR_COMPILER=0 UNRELATED_FLAG=7)
   if(NOT _required IN_LIST _runtime)
      message(FATAL_ERROR "Missing runtime profile definition: ${_required}")
   endif()
endforeach()
forge_wamr_apply_profile("${CMAKE_CURRENT_SOURCE_DIR}" compiler)
get_directory_property(_compiler COMPILE_DEFINITIONS)
list(REMOVE_ITEM _runtime WASM_ENABLE_WAMR_COMPILER=0)
list(REMOVE_ITEM _compiler WASM_ENABLE_WAMR_COMPILER=1)
if(FORGE_ENABLE_VM_WASM_LLVM_JIT)
   if(NOT "WASM_ENABLE_JIT=1" IN_LIST _runtime OR NOT "WASM_ENABLE_JIT=0" IN_LIST _compiler)
      message(FATAL_ERROR "Runtime JIT selection leaked into the isolated AOT compiler")
   endif()
   list(REMOVE_ITEM _runtime WASM_ENABLE_JIT=1)
   list(REMOVE_ITEM _compiler WASM_ENABLE_JIT=0)
endif()
if(NOT _runtime STREQUAL _compiler)
   message(FATAL_ERROR "Compiler/runtime profile differs beyond their compiler/JIT roles")
endif()
set(_unique ${_compiler})
list(TRANSFORM _unique REPLACE "=.*$" "")
list(LENGTH _unique _count)
list(REMOVE_DUPLICATES _unique)
list(LENGTH _unique _unique_count)
if(NOT _count EQUAL _unique_count)
   message(FATAL_ERROR "Profile retained conflicting upstream definitions")
endif()
set(WAMR_BUILD_FAST_INTERP ON)
forge_wamr_apply_profile("${CMAKE_CURRENT_SOURCE_DIR}" runtime)
get_directory_property(_fast COMPILE_DEFINITIONS)
if(NOT "WASM_ENABLE_FAST_INTERP=1" IN_LIST _fast)
   message(FATAL_ERROR "Explicit standalone fast-mode qualification was lost")
endif()

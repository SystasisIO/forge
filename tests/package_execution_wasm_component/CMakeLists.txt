cmake_minimum_required(VERSION 3.31)
project(forge-package-execution-wasm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Forge CONFIG REQUIRED COMPONENTS execution_wasm)
get_target_property(dependencies Forge::forge_execution_wasm INTERFACE_LINK_LIBRARIES)
if(NOT "${dependencies}" STREQUAL "Forge::forge_vm_wasm_interpret;Forge::forge_chain_protocol;Forge::forge_raw;Forge::forge_exceptions")
   message(FATAL_ERROR "Executor dependency boundary changed: ${dependencies}")
endif()
add_executable(forge_package_execution_wasm_component main.cpp)
target_link_libraries(forge_package_execution_wasm_component PRIVATE Forge::forge_execution_wasm)

cmake_minimum_required(VERSION 3.31)
project(forge-package-swarm-storage-component LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_DISABLE_FIND_PACKAGE_Ngtcp2 ON)
set(CMAKE_DISABLE_FIND_PACKAGE_Notcurses ON)
set(CMAKE_DISABLE_FIND_PACKAGE_glaze ON)
find_package(Forge CONFIG REQUIRED COMPONENTS swarm_storage)
get_target_property(_links Forge::forge_swarm_storage INTERFACE_LINK_LIBRARIES)
if(_links MATCHES "forge_(chain|db|api|app|net|plugins)(_|;|$)")
   message(FATAL_ERROR "File storage acquired an unrelated runtime dependency: ${_links}")
endif()
add_executable(forge_package_swarm_storage_component main.cpp)
target_link_libraries(forge_package_swarm_storage_component PRIVATE Forge::forge_swarm_storage)

cmake_minimum_required(VERSION 3.31)
project(forge-package-swarm-storage-component LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_DISABLE_FIND_PACKAGE_Ngtcp2 ON)
set(CMAKE_DISABLE_FIND_PACKAGE_Notcurses ON)
set(CMAKE_DISABLE_FIND_PACKAGE_glaze ON)
find_package(Forge CONFIG REQUIRED COMPONENTS swarm_storage)
get_target_property(_links Forge::forge_swarm_storage INTERFACE_LINK_LIBRARIES)
if(_links MATCHES "forge_(chain|api|app|net|plugins|rocksdb)(_|;|$)")
   message(FATAL_ERROR "Swarm storage acquired an unrelated runtime dependency: ${_links}")
endif()
foreach(_link IN LISTS _links)
   if(_link MATCHES "forge_db_" AND NOT _link MATCHES "^Forge::forge_db_(core|object|ids)$")
      message(FATAL_ERROR "Swarm storage selected a concrete or unrelated DB layer: ${_link}")
   endif()
endforeach()
add_executable(forge_package_swarm_storage_component main.cpp)
target_link_libraries(forge_package_swarm_storage_component PRIVATE Forge::forge_swarm_storage)

cmake_minimum_required(VERSION 3.31)
project(forge_wamr_load_budget_probe LANGUAGES C CXX ASM)
enable_testing()
find_package(Threads REQUIRED)
get_filename_component(_forge_root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(FORGE_WAMR_LOAD_BUDGET_SOURCE "" CACHE PATH "Explicit unpublished fork checkout, isolated qualification only")
set(FORGE_WAMR_LOAD_BUDGET_WASM "" CACHE FILEPATH "Existing source fixture")
set(FORGE_WAMR_LOAD_BUDGET_AOT "" CACHE FILEPATH "Existing native fixture")
option(FORGE_WAMR_LOAD_BUDGET_SANITIZERS "Sanitize the real loader and private owner" OFF)
if(NOT EXISTS "${FORGE_WAMR_LOAD_BUDGET_SOURCE}/build-scripts/runtime_lib.cmake")
   message(FATAL_ERROR "Explicit fork source is required; this harness never changes the production pin")
endif()
foreach(_fixture IN ITEMS FORGE_WAMR_LOAD_BUDGET_WASM FORGE_WAMR_LOAD_BUDGET_AOT)
   if(NOT EXISTS "${${_fixture}}")
      message(FATAL_ERROR "Supply existing ${_fixture}; no compiler build or archive download is performed")
   endif()
endforeach()
include("${_forge_root}/cmake/ForgeWamrProfile.cmake")
string(TOLOWER "${CMAKE_SYSTEM_NAME}" WAMR_BUILD_PLATFORM)
if(APPLE)
   add_compile_definitions(BH_PLATFORM_DARWIN)
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
   set(WAMR_BUILD_TARGET AARCH64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
   set(WAMR_BUILD_TARGET X86_64)
else()
   message(FATAL_ERROR "No qualified bounded-map platform: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
set(WAMR_ROOT_DIR "${FORGE_WAMR_LOAD_BUDGET_SOURCE}")
include("${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake")
forge_wamr_apply_profile("${CMAKE_CURRENT_SOURCE_DIR}" runtime)
add_library(wamr_load_budget_runtime STATIC ${WAMR_RUNTIME_LIB_SOURCE})
target_link_libraries(wamr_load_budget_runtime PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
set(_interpret "${_forge_root}/libraries/vm/wasm/interpret")
add_executable(wamr_load_budget_probe ../load_budget_probe.cpp
   "${_interpret}/engine_load_budget.cpp" "${_interpret}/engine_runtime.cpp"
   "${_interpret}/engine_memory.cpp" "${_interpret}/engine_module.cpp"
   "${_interpret}/engine_module_jit.cpp"
   "${_interpret}/engine_bindings.cpp")
target_compile_features(wamr_load_budget_probe PRIVATE cxx_std_23)
target_compile_options(wamr_load_budget_probe PRIVATE -Wall -Wextra -Wpedantic -Werror
   -fno-fast-math -ffp-contract=off -frounding-math)
target_link_libraries(wamr_load_budget_probe PRIVATE wamr_load_budget_runtime)
if(FORGE_WAMR_LOAD_BUDGET_SANITIZERS)
   foreach(_target IN ITEMS wamr_load_budget_runtime wamr_load_budget_probe)
      target_compile_options(${_target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
   endforeach()
   target_link_options(wamr_load_budget_probe PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME wamr_load_budget_real_loader COMMAND wamr_load_budget_probe
   "${FORGE_WAMR_LOAD_BUDGET_WASM}" "${FORGE_WAMR_LOAD_BUDGET_AOT}")
set_tests_properties(wamr_load_budget_real_loader PROPERTIES TIMEOUT 60 LABELS "wamr;preparation;private")

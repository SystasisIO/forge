cmake_minimum_required(VERSION 3.31)

project(forge-package-metadata LANGUAGES CXX)

find_package(Forge CONFIG REQUIRED COMPONENTS core)

if(NOT TARGET Forge::forge_core)
   message(FATAL_ERROR "Focused package discovery did not expose Forge::forge_core")
endif()
if(NOT Forge_NOTICES_DIR)
   message(FATAL_ERROR "Forge package does not expose its legal-notices metadata")
endif()
if(DEFINED FORGE_EXPECTED_NOTICES_DIR AND NOT Forge_NOTICES_DIR STREQUAL FORGE_EXPECTED_NOTICES_DIR)
   message(FATAL_ERROR "Forge notices path mismatch: ${Forge_NOTICES_DIR}; expected ${FORGE_EXPECTED_NOTICES_DIR}")
endif()
if(NOT Forge_NOTICE_FILES)
   message(FATAL_ERROR "Forge package does not expose its legal-notices file list")
endif()
set(notice_names)
foreach(notice IN LISTS Forge_NOTICE_FILES)
   if(notice STREQUAL "." OR notice STREQUAL ".." OR notice MATCHES "[/\\\\]" OR notice IN_LIST notice_names)
      message(FATAL_ERROR "Legal-notices file list contains a non-basename or duplicate: ${notice}")
   endif()
   list(APPEND notice_names "${notice}")
endforeach()
foreach(required IN ITEMS LICENSE NOTICE AUTHORS PROVENANCE.md THIRD_PARTY_LICENSES LICENSE.eos-vm)
   if(NOT required IN_LIST Forge_NOTICE_FILES)
      message(FATAL_ERROR "Legal-notices file list omits ${required}")
   endif()
endforeach()
foreach(optional IN ITEMS LICENSE.Spring LICENSE.WAVM)
   if(Forge_BUILT_WITH_VM_WASM_OPTIMIZING AND NOT optional IN_LIST Forge_NOTICE_FILES)
      message(FATAL_ERROR "Optimizing compiler legal-notices file list omits ${optional}")
   elseif(NOT Forge_BUILT_WITH_VM_WASM_OPTIMIZING AND optional IN_LIST Forge_NOTICE_FILES)
      message(FATAL_ERROR "Disabled optimizing compiler unexpectedly advertises ${optional}")
   endif()
endforeach()
if(FORGE_EXPECTED_NOTICE_METADATA)
   include("${FORGE_EXPECTED_NOTICE_METADATA}")
   if(NOT Forge_NOTICE_FILES STREQUAL FORGE_PACKAGE_NOTICE_FILES)
      message(FATAL_ERROR "Legal-notices file list differs from the owning install-source list")
   endif()
endif()
if(FORGE_EXPECT_NOTICES_PRESENT)
   foreach(notice IN LISTS Forge_NOTICE_FILES)
      if(NOT EXISTS "${Forge_NOTICES_DIR}/${notice}" OR IS_DIRECTORY "${Forge_NOTICES_DIR}/${notice}")
         message(FATAL_ERROR "Legal-notices metadata does not locate ${notice}")
      endif()
   endforeach()
endif()

cmake_minimum_required(VERSION 3.31)
project(forge-package-auth-liboauth2-component LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(FORGE_TEST_ATTESTATION_LOOKUPS "Check Core/JOSE/OAuth lookup ordering with a JOSE-enabled producer" OFF)
if(FORGE_TEST_ATTESTATION_LOOKUPS)
   find_package(Forge CONFIG REQUIRED COMPONENTS core)
   if(TARGET cjose::cjose_static)
      message(FATAL_ERROR "Core-only lookup must not discover the JOSE backend")
   endif()
   find_package(Forge CONFIG REQUIRED COMPONENTS crypto_jose)
endif()
find_package(Forge CONFIG REQUIRED COMPONENTS auth_liboauth2)
if(FORGE_TEST_ATTESTATION_LOOKUPS)
   get_target_property(_jose_links_before Forge::forge_crypto_jose INTERFACE_LINK_LIBRARIES)
   get_target_property(_oauth_links_before Forge::oauth2_internal INTERFACE_LINK_LIBRARIES)
   if(NOT "$<LINK_ONLY:cjose::cjose_static>" IN_LIST _jose_links_before
      OR NOT "cjose::cjose_static" IN_LIST _oauth_links_before)
      message(FATAL_ERROR "The original JOSE private and OAuth public backend edges must be retained")
   endif()
   find_package(Forge CONFIG REQUIRED COMPONENTS crypto_jose auth_liboauth2)
   get_target_property(_jose_links_after Forge::forge_crypto_jose INTERFACE_LINK_LIBRARIES)
   get_target_property(_oauth_links_after Forge::oauth2_internal INTERFACE_LINK_LIBRARIES)
   if(NOT _jose_links_before STREQUAL _jose_links_after OR NOT _oauth_links_before STREQUAL _oauth_links_after)
      message(FATAL_ERROR "Repeated component lookup changed a qualified backend binding")
   endif()
endif()

add_executable(forge_package_auth_liboauth2_component main.cpp)
target_link_libraries(forge_package_auth_liboauth2_component PRIVATE Forge::forge_auth_liboauth2)

cmake_minimum_required(VERSION 3.31)
project(forge-package-expression-component LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT FORGE_EXPECTED_PREFIX)
   message(FATAL_ERROR "FORGE_EXPECTED_PREFIX is required for installed-package acceptance")
endif()
find_package(Forge CONFIG REQUIRED COMPONENTS expression)

file(REAL_PATH "${FORGE_EXPECTED_PREFIX}" expected_prefix)
file(REAL_PATH "${Forge_DIR}" package_directory)
cmake_path(IS_PREFIX expected_prefix "${package_directory}" NORMALIZE package_is_relocated)
if(NOT package_is_relocated OR NOT Forge_BUILT_WITH_EXPRESSION)
   message(FATAL_ERROR "The consumer did not discover the relocated expression package")
endif()

get_target_property(worker_executable ForgeCEL::worker IMPORTED_LOCATION)
file(REAL_PATH "${worker_executable}" worker_executable)
cmake_path(IS_PREFIX expected_prefix "${worker_executable}" NORMALIZE worker_is_relocated)
get_target_property(worker_type ForgeCEL::worker TYPE)
if(NOT worker_is_relocated OR NOT EXISTS "${worker_executable}" OR NOT worker_type STREQUAL "EXECUTABLE")
   message(FATAL_ERROR "The CEL worker executable is not supplied by the relocated package")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/worker-location.txt" CONTENT "${worker_executable}\n")
get_target_property(expression_includes Forge::forge_expression INTERFACE_INCLUDE_DIRECTORIES)
if("${expression_includes}" MATCHES "(cel-cpp|protobuf|abseil|/absl/|/vendor/)")
   message(FATAL_ERROR "Expression exposes private dependency headers: ${expression_includes}")
endif()
get_target_property(expression_links Forge::forge_expression INTERFACE_LINK_LIBRARIES)
if("${expression_links}" MATCHES "(ForgeCEL::backend|protobuf|abseil|absl::)")
   message(FATAL_ERROR "The consumer links the private CEL runtime instead of using its worker")
endif()

add_executable(forge_package_expression_component main.cpp)
target_link_libraries(forge_package_expression_component PRIVATE Forge::forge_expression)
target_compile_definitions(forge_package_expression_component PRIVATE
   "FORGE_EXPRESSION_WORKER=\"${worker_executable}\"")
if(APPLE)
   target_link_options(forge_package_expression_component PRIVATE
      "LINKER:-map,${CMAKE_CURRENT_BINARY_DIR}/expression.linkmap")
endif()

cmake_minimum_required(VERSION 3.31)
project(forge-package-swarm-db-component LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_DISABLE_FIND_PACKAGE_Ngtcp2 ON)
set(CMAKE_DISABLE_FIND_PACKAGE_Notcurses ON)
find_package(Forge CONFIG REQUIRED COMPONENTS swarm_db)
get_target_property(_links Forge::forge_swarm_db INTERFACE_LINK_LIBRARIES)
if(_links MATCHES "forge_(swarm_storage|net|plugins|db_(rocksdb|mdbx|blob))")
   message(FATAL_ERROR "Swarm models acquired a concrete runtime: ${_links}")
endif()
add_executable(forge_package_swarm_db_component main.cpp)
target_link_libraries(forge_package_swarm_db_component PRIVATE Forge::forge_swarm_db)

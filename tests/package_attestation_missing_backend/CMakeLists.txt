cmake_minimum_required(VERSION 3.31)
project(ForgeAttestationMissingBackend LANGUAGES CXX)
option(FORGE_ATTESTATION_MISSING_JANSSON
   "Exercise missing Jansson; OFF requires an unpatched external CJOSE" ON)
if(FORGE_ATTESTATION_MISSING_JANSSON)
   set(CMAKE_DISABLE_FIND_PACKAGE_Jansson TRUE)
endif()
find_package(Forge CONFIG QUIET COMPONENTS crypto_jose)
if(NOT DEFINED Forge_FOUND OR Forge_FOUND)
   message(FATAL_ERROR "Missing JOSE backend dependency must fail optional package lookup")
endif()
if(NOT Forge_NOT_FOUND_MESSAGE MATCHES "dependency (cjose|CJOSE)")
   message(FATAL_ERROR "JOSE package lookup failed for an unrelated reason")
endif()
if(NOT FORGE_ATTESTATION_MISSING_JANSSON)
   if(NOT cjose_FOUND OR NOT TARGET cjose::cjose_static)
      message(FATAL_ERROR "Unpatched backend rejection requires the actual native CJOSE static package")
   endif()
   if(NOT DEFINED FORGE_CJOSE_HAS_QUALIFIED_DEPENDENCIES
      OR NOT FORGE_CJOSE_HAS_QUALIFIED_DEPENDENCIES)
      message(FATAL_ERROR "Qualified Jansson dependency probe must execute and pass")
   endif()
   if(NOT DEFINED FORGE_CJOSE_HAS_EXTERNAL_SIGNING OR FORGE_CJOSE_HAS_EXTERNAL_SIGNING)
      message(FATAL_ERROR "Unpatched CJOSE extension probe must execute and fail")
   endif()
endif()

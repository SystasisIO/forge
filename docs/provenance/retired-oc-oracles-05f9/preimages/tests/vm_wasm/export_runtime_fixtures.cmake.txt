cmake_minimum_required(VERSION 3.31)

# One CTest invocation owns this tree; no concurrent invocation shares receipt.
# Historical generation directories are never removed.
function(forge_oc_absolute variable)
   if(NOT DEFINED ${variable} OR NOT IS_ABSOLUTE "${${variable}}" OR
      "${${variable}}" MATCHES "[;\r\n]" OR IS_SYMLINK "${${variable}}")
      message(FATAL_ERROR "${variable} must be an absolute non-symlink owner path")
   endif()
endfunction()

function(forge_oc_source_identity output revision)
   forge_oc_absolute(FORGE_OC_SOURCE_ROOT)
   # Bounded owning source scope, not a parallel CMake dependency graph or a
   # security attestation of external SDK/toolchain/system shared libraries.
   set(sources "")
   foreach(owner IN ITEMS libraries/tooling/wasm_compiler libraries/vm/wasm/optimizing
      libraries/vm/wasm/interpret libraries/raw libraries/core libraries/exceptions vendor/softfloat)
      foreach(extension IN ITEMS cpp cppm h hpp hxx ipp c inc S s)
         file(GLOB_RECURSE files LIST_DIRECTORIES FALSE "${FORGE_OC_SOURCE_ROOT}/${owner}/*.${extension}")
         list(APPEND sources ${files})
      endforeach()
      file(GLOB_RECURSE files LIST_DIRECTORIES FALSE "${FORGE_OC_SOURCE_ROOT}/${owner}/CMakeLists.txt")
      list(APPEND sources ${files})
   endforeach()
   list(APPEND sources
      "${FORGE_OC_SOURCE_ROOT}/CMakeLists.txt"
      "${FORGE_OC_SOURCE_ROOT}/cmake/ForgeSoftFloatAbi.cmake"
      "${FORGE_OC_SOURCE_ROOT}/cmake/ForgeVendorCompilePolicy.cmake"
      "${FORGE_OC_SOURCE_ROOT}/tests/vm_wasm/CMakeLists.txt"
      "${FORGE_OC_SOURCE_ROOT}/tests/vm_wasm/main.cpp"
      "${FORGE_OC_SOURCE_ROOT}/tests/vm_wasm/test_prelude.hpp"
      "${FORGE_OC_SOURCE_ROOT}/tests/vm_wasm/fresh_process.hpp"
      "${FORGE_OC_SOURCE_ROOT}/tests/vm_wasm/optimizing_fixture_export.hpp"
      "${FORGE_OC_SOURCE_ROOT}/tests/vm_wasm/optimizing_compiler_tests.cpp"
      "${FORGE_OC_SOURCE_ROOT}/tests/vm_wasm/optimizing_epoch_ir_tests.cpp"
      "${CMAKE_CURRENT_FUNCTION_LIST_FILE}")
   list(REMOVE_DUPLICATES sources)
   list(SORT sources)
   list(LENGTH sources count)
   if(count GREATER 2048)
      message(FATAL_ERROR "fixture source scope exceeds bounded file count")
   endif()
   set(total 0)
   set(binding "")
   foreach(source IN LISTS sources)
      if(NOT EXISTS "${source}" OR IS_DIRECTORY "${source}" OR IS_SYMLINK "${source}")
         message(FATAL_ERROR "fixture source unavailable: ${source}")
      endif()
      file(SIZE "${source}" size)
      math(EXPR total "${total} + ${size}")
      if(total GREATER 67108864)
         message(FATAL_ERROR "fixture source scope exceeds bounded 64MiB input")
      endif()
      file(SHA256 "${source}" digest)
      file(RELATIVE_PATH relative "${FORGE_OC_SOURCE_ROOT}" "${source}")
      string(APPEND binding "${relative}:${digest}\n")
   endforeach()
   string(SHA256 digest "${binding}")
   execute_process(COMMAND git -C "${FORGE_OC_SOURCE_ROOT}" rev-parse HEAD
      RESULT_VARIABLE result OUTPUT_VARIABLE head OUTPUT_STRIP_TRAILING_WHITESPACE TIMEOUT 5)
   string(LENGTH "${head}" head_size)
   if(NOT result STREQUAL "0" OR NOT head_size EQUAL 40 OR NOT head MATCHES "^[0-9a-f]+$")
      message(FATAL_ERROR "fixture source revision unavailable")
   endif()
   set(${output} "${digest}" PARENT_SCOPE)
   set(${revision} "${head}" PARENT_SCOPE)
endfunction()

function(forge_oc_build_identity output)
   forge_oc_absolute(FORGE_OC_BUILD_CONFIG)
   if(NOT EXISTS "${FORGE_OC_BUILD_CONFIG}" OR IS_DIRECTORY "${FORGE_OC_BUILD_CONFIG}")
      message(FATAL_ERROR "configured compiler target identity missing; configure normal root first")
   endif()
   file(SIZE "${FORGE_OC_BUILD_CONFIG}" size)
   if(size GREATER 16384)
      message(FATAL_ERROR "compiler build configuration exceeds bounded input")
   endif()
   file(SHA256 "${FORGE_OC_BUILD_CONFIG}" digest)
   set(${output} "${digest}" PARENT_SCOPE)
endfunction()

function(forge_oc_bind_build)
   # This mode belongs ONLY to the existing compiler target's POST_BUILD.
   # The tree/inputs are frozen throughout that owning build. It is never
   # invoked by setup/consumers to bless a stale executable.
   forge_oc_source_identity(source revision)
   forge_oc_build_identity(configuration)
   forge_oc_absolute(FORGE_OC_COMPILER)
   forge_oc_absolute(FORGE_OC_BUILD_STAMP)
   file(SHA256 "${FORGE_OC_COMPILER}" compiler)
   set(stamp "{}")
   foreach(field IN ITEMS source revision configuration compiler)
      string(JSON stamp SET "${stamp}" "${field}" "\"${${field}}\"")
   endforeach()
   string(JSON stamp SET "${stamp}" version 1)
   file(WRITE "${FORGE_OC_BUILD_STAMP}.tmp" "${stamp}\n")
   file(RENAME "${FORGE_OC_BUILD_STAMP}.tmp" "${FORGE_OC_BUILD_STAMP}")
endfunction()

function(forge_oc_verify_build output)
   forge_oc_absolute(FORGE_OC_BUILD_STAMP)
   if(NOT EXISTS "${FORGE_OC_BUILD_STAMP}" OR IS_DIRECTORY "${FORGE_OC_BUILD_STAMP}")
      message(FATAL_ERROR
         "compiler build stamp missing: cmake --build <this build tree> --target test_forge_vm_wasm_optimizing_compiler -j4 before CTest")
   endif()
   file(SIZE "${FORGE_OC_BUILD_STAMP}" size)
   if(size GREATER 4096)
      message(FATAL_ERROR "compiler target stamp exceeds bounded input")
   endif()
   file(READ "${FORGE_OC_BUILD_STAMP}" stamp)
   string(JSON version GET "${stamp}" version)
   if(NOT version EQUAL 1)
      message(FATAL_ERROR "compiler target stamp unsupported")
   endif()
   forge_oc_source_identity(source revision)
   forge_oc_build_identity(configuration)
   forge_oc_absolute(FORGE_OC_COMPILER)
   file(SHA256 "${FORGE_OC_COMPILER}" compiler)
   foreach(field IN ITEMS source revision configuration compiler)
      string(JSON expected GET "${stamp}" "${field}")
      if(NOT expected STREQUAL "${${field}}")
         message(FATAL_ERROR
            "stale compiler target binding. With the supported Ninja graph and frozen inputs, remove ONLY its exact owned POST_BUILD stamp:\ncmake -E rm -f \"${FORGE_OC_BUILD_STAMP}\"\nThen rebuild SAME target, without --clean-first:\ncmake --build \"${FORGE_OC_BUILD_ROOT}\" --target test_forge_vm_wasm_optimizing_compiler -j4\nCTest never performs either operation.")
      endif()
   endforeach()
   file(SHA256 "${FORGE_OC_BUILD_STAMP}" digest)
   set(${output} "${digest}" PARENT_SCOPE)
endfunction()

function(forge_oc_fixture_names output)
   set(names "")
   foreach(profile IN ITEMS darwin-arm64 linux-x86-64)
      foreach(fixture IN ITEMS add imports-indirect grow integer-traps memory-load import-loop finite recursive infinite)
         foreach(extension IN ITEMS request foc)
            list(APPEND names "${fixture}-${profile}.${extension}")
         endforeach()
      endforeach()
   endforeach()
   list(SORT names)
   set(${output} "${names}" PARENT_SCOPE)
endfunction()

function(forge_oc_fixture_owner output)
   forge_oc_absolute(FORGE_OC_BUILD_ROOT)
   file(REAL_PATH "${FORGE_OC_BUILD_ROOT}" build)
   if(NOT IS_DIRECTORY "${build}" OR build STREQUAL FORGE_OC_SOURCE_ROOT)
      message(FATAL_ERROR "fixtures require a separate existing build tree")
   endif()
   set(root "${build}/tests/vm_wasm/optimizing-fixtures")
   foreach(parent IN ITEMS "${build}/tests" "${build}/tests/vm_wasm")
      if(IS_SYMLINK "${parent}")
         message(FATAL_ERROR "fixture output ancestor is a symlink")
      endif()
   endforeach()
   if(IS_SYMLINK "${root}")
      message(FATAL_ERROR "fixture owner root is a symlink")
   endif()
   if(EXISTS "${root}")
      file(REAL_PATH "${root}" resolved)
      if(NOT resolved STREQUAL root OR NOT IS_DIRECTORY "${root}" OR
         NOT EXISTS "${root}/owner.txt" OR IS_SYMLINK "${root}/owner.txt")
         message(FATAL_ERROR "fixture output ownership unavailable")
      endif()
      file(READ "${root}/owner.txt" owner LIMIT 4096)
      if(NOT owner STREQUAL "forge-oc-fixtures-v1\n${FORGE_OC_BUILD_ROOT}\n")
         message(FATAL_ERROR "fixture owner marker mismatch")
      endif()
   endif()
   set(${output} "${root}" PARENT_SCOPE)
endfunction()

function(forge_oc_new_generation output)
   forge_oc_fixture_owner(root)
   set(owner "${root}/owner.txt")
   set(expected "forge-oc-fixtures-v1\n${FORGE_OC_BUILD_ROOT}\n")
   if(EXISTS "${root}")
      if(NOT IS_DIRECTORY "${root}" OR NOT EXISTS "${owner}" OR IS_SYMLINK "${owner}")
         message(FATAL_ERROR "fixture output has no ownership marker")
      endif()
      file(READ "${owner}" actual LIMIT 4096)
      if(NOT actual STREQUAL expected)
         message(FATAL_ERROR "fixture output owner mismatch")
      endif()
   else()
      file(MAKE_DIRECTORY "${root}")
      file(CHMOD "${root}" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE)
      file(WRITE "${owner}" "${expected}")
   endif()
   # Only this known small receipt is invalidated. Never remove supplied
   # directories, partial output or previous successful payloads.
   set(receipt "${root}/current.json")
   if(IS_SYMLINK "${receipt}" OR IS_DIRECTORY "${receipt}")
      message(FATAL_ERROR "fixture receipt is not an owned regular file")
   endif()
   file(REMOVE "${receipt}")
   foreach(attempt RANGE 1 8)
      string(RANDOM LENGTH 20 ALPHABET 0123456789abcdef generation)
      set(run "${root}/run-${generation}")
      if(NOT EXISTS "${run}" AND NOT IS_SYMLINK "${run}")
         file(MAKE_DIRECTORY "${run}/payload")
         file(CHMOD "${run}" "${run}/payload" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE)
         set(${output} "${run}" PARENT_SCOPE)
         return()
      endif()
   endforeach()
   message(FATAL_ERROR "cannot allocate a fresh fixture generation")
endfunction()

function(forge_oc_setup)
   forge_oc_new_generation(run) # invalidate only the owned receipt even on later preflight failure
   forge_oc_verify_build(build_binding) # refusal, never a hidden CTest build
   forge_oc_absolute(FORGE_OC_COMPILER)
   if(NOT EXISTS "${FORGE_OC_COMPILER}" OR IS_DIRECTORY "${FORGE_OC_COMPILER}")
      message(FATAL_ERROR "build the real compiler test target before setup")
   endif()
   foreach(selector IN ITEMS optimizing_compiler/export_checked_runtime_fixtures
      optimizing_epoch_ir/export_checked_infinite_runtime_fixture)
      string(REPLACE "/" "-" label "${selector}")
      execute_process(COMMAND "${CMAKE_COMMAND}" -E env
         "FORGE_OC_FIXTURE_EXPORT_DIRECTORY=${run}/payload"
         "${FORGE_OC_COMPILER}" --catch_system_errors=no "--run_test=${selector}"
         RESULT_VARIABLE result TIMEOUT 60
         OUTPUT_FILE "${run}/${label}.log" ERROR_FILE "${run}/${label}.err")
      if(NOT result STREQUAL "0")
         message(FATAL_ERROR "fresh export failed (${result}); partial output retained: ${run}")
      endif()
   endforeach()
   forge_oc_fixture_names(names)
   file(GLOB actual LIST_DIRECTORIES TRUE RELATIVE "${run}/payload" "${run}/payload/*")
   list(SORT actual)
   if(NOT actual STREQUAL names)
      message(FATAL_ERROR "fresh export must contain exactly 18 request/artifact pairs")
   endif()
   set(files "{}")
   foreach(name IN LISTS names)
      set(path "${run}/payload/${name}")
      file(SIZE "${path}" size)
      if(IS_SYMLINK "${path}" OR IS_DIRECTORY "${path}" OR size LESS 1 OR size GREATER 2097152)
         message(FATAL_ERROR "fixture outside bounded regular-file input")
      endif()
      file(SHA256 "${path}" digest)
      string(JSON files SET "${files}" "${name}" "\"${digest}\"")
      # An opt-in reviewed reference is checked, never used instead of fresh
      # output. Existing recognized files must match; no supplied file is changed.
      if(DEFINED FORGE_OC_REVIEWED_DIRECTORY AND NOT FORGE_OC_REVIEWED_DIRECTORY STREQUAL "")
         forge_oc_absolute(FORGE_OC_REVIEWED_DIRECTORY)
         if(EXISTS "${FORGE_OC_REVIEWED_DIRECTORY}/${name}")
            file(SIZE "${FORGE_OC_REVIEWED_DIRECTORY}/${name}" reviewed_size)
            if(IS_SYMLINK "${FORGE_OC_REVIEWED_DIRECTORY}/${name}" OR
               IS_DIRECTORY "${FORGE_OC_REVIEWED_DIRECTORY}/${name}" OR
               reviewed_size LESS 1 OR reviewed_size GREATER 2097152)
               message(FATAL_ERROR "reviewed fixture outside bounded regular-file input")
            endif()
            file(SHA256 "${FORGE_OC_REVIEWED_DIRECTORY}/${name}" reviewed)
            if(NOT reviewed STREQUAL digest)
               message(FATAL_ERROR "reviewed reference differs: ${name}")
            endif()
         endif()
      endif()
   endforeach()
   if(DEFINED FORGE_OC_REVIEWED_DIRECTORY AND NOT FORGE_OC_REVIEWED_DIRECTORY STREQUAL "")
      foreach(profile IN ITEMS darwin-arm64 linux-x86-64)
         foreach(extension IN ITEMS request foc)
            if(NOT EXISTS "${FORGE_OC_REVIEWED_DIRECTORY}/add-${profile}.${extension}")
               message(FATAL_ERROR "reviewed reference needs both add pairs")
            endif()
         endforeach()
      endforeach()
   endif()
   forge_oc_verify_build(after_export_binding)
   if(NOT after_export_binding STREQUAL build_binding)
      message(FATAL_ERROR "compiler build binding changed during export")
   endif()
   forge_oc_source_identity(source revision)
   file(SHA256 "${FORGE_OC_COMPILER}" compiler)
   string(TIMESTAMP created "%s" UTC)
   # JSON is a test-only path/digest receipt, never executable include()/codec.
   set(manifest "{}")
   foreach(field IN ITEMS run source revision compiler created build_binding)
      string(REPLACE "\\" "\\\\" value "${${field}}")
      string(REPLACE "\"" "\\\"" value "${value}")
      string(JSON manifest SET "${manifest}" "${field}" "\"${value}\"")
   endforeach()
   string(JSON manifest SET "${manifest}" version 1)
   string(JSON manifest SET "${manifest}" files "${files}")
   forge_oc_fixture_owner(root)
   file(WRITE "${run}/manifest.json" "${manifest}\n")
   file(RENAME "${run}/manifest.json" "${root}/current.json")
   message(STATUS "Fresh verified 18-pair fixture generation: ${run}/payload")
endfunction()

function(forge_oc_verify_fixtures output)
   forge_oc_verify_build(build_binding)
   forge_oc_fixture_owner(root)
   set(receipt "${root}/current.json")
   if(NOT EXISTS "${receipt}" OR IS_SYMLINK "${receipt}" OR IS_DIRECTORY "${receipt}")
      message(FATAL_ERROR "fresh setup receipt missing")
   endif()
   file(SIZE "${receipt}" size)
   if(size GREATER 16384)
      message(FATAL_ERROR "fixture receipt exceeds bounded input")
   endif()
   file(READ "${receipt}" manifest)
   string(JSON version GET "${manifest}" version)
   string(JSON run GET "${manifest}" run)
   string(JSON created GET "${manifest}" created)
   string(TIMESTAMP now "%s" UTC)
   math(EXPR age "${now} - ${created}")
   if(NOT version EQUAL 1 OR age LESS 0 OR age GREATER 3600)
      message(FATAL_ERROR "fixture generation expired/unsupported")
   endif()
   get_filename_component(parent "${run}" DIRECTORY)
   get_filename_component(name "${run}" NAME)
   if(NOT parent STREQUAL root OR NOT name MATCHES "^run-[0-9a-f]+$" OR
      IS_SYMLINK "${run}" OR IS_SYMLINK "${run}/payload" OR NOT IS_DIRECTORY "${run}/payload")
      message(FATAL_ERROR "fixture generation outside its exact owner")
   endif()
   forge_oc_source_identity(source revision)
   forge_oc_absolute(FORGE_OC_COMPILER)
   file(SHA256 "${FORGE_OC_COMPILER}" compiler)
   foreach(field IN ITEMS source revision compiler build_binding)
      string(JSON expected GET "${manifest}" "${field}")
      if(NOT expected STREQUAL "${${field}}")
         message(FATAL_ERROR "fixture does not match current source/compiler")
      endif()
   endforeach()
   forge_oc_fixture_names(names)
   file(GLOB actual LIST_DIRECTORIES TRUE RELATIVE "${run}/payload" "${run}/payload/*")
   list(SORT actual)
   string(JSON count LENGTH "${manifest}" files)
   if(NOT actual STREQUAL names OR NOT count EQUAL 36)
      message(FATAL_ERROR "fixture payload set changed")
   endif()
   foreach(name IN LISTS names)
      set(path "${run}/payload/${name}")
      if(IS_SYMLINK "${path}" OR IS_DIRECTORY "${path}")
         message(FATAL_ERROR "fixture is not an owned regular file")
      endif()
      file(SIZE "${path}" size)
      if(size LESS 1 OR size GREATER 2097152)
         message(FATAL_ERROR "fixture length changed")
      endif()
      file(SHA256 "${path}" digest)
      string(JSON expected GET "${manifest}" files "${name}")
      if(NOT expected STREQUAL digest)
         message(FATAL_ERROR "fixture digest changed: ${name}")
      endif()
   endforeach()
   set(${output} "${run}/payload" PARENT_SCOPE)
endfunction()

function(forge_oc_native_suffix output)
   execute_process(COMMAND uname -s OUTPUT_VARIABLE system OUTPUT_STRIP_TRAILING_WHITESPACE
      RESULT_VARIABLE system_result TIMEOUT 5)
   execute_process(COMMAND uname -m OUTPUT_VARIABLE processor OUTPUT_STRIP_TRAILING_WHITESPACE
      RESULT_VARIABLE processor_result TIMEOUT 5)
   if(NOT system_result STREQUAL "0" OR NOT processor_result STREQUAL "0")
      message(FATAL_ERROR "native host identification failed")
   endif()
   if(system STREQUAL "Darwin" AND processor STREQUAL "arm64")
      set(value darwin-arm64)
   elseif(system STREQUAL "Linux" AND processor STREQUAL "x86_64")
      set(value linux-x86-64)
   else()
      message(FATAL_ERROR "requires native Darwin arm64 or Linux x86-64")
   endif()
   set(${output} "${value}" PARENT_SCOPE)
endfunction()

if(DEFINED FORGE_OC_MODE)
   if(FORGE_OC_MODE STREQUAL "bind")
      forge_oc_bind_build()
   elseif(FORGE_OC_MODE STREQUAL "setup")
      forge_oc_setup()
   elseif(FORGE_OC_MODE STREQUAL "run")
      forge_oc_verify_fixtures(directory)
      forge_oc_absolute(FORGE_OC_CONSUMER)
      if(FORGE_OC_GATE STREQUAL "executor")
         execute_process(COMMAND "${CMAKE_COMMAND}" -E env "FORGE_OC_RUNTIME_FIXTURE_DIRECTORY=${directory}"
            "${FORGE_OC_CONSUMER}" --catch_system_errors=no "--run_test=optimizing_executor_compiled/*"
            --report_level=detailed RESULT_VARIABLE result TIMEOUT 60 OUTPUT_VARIABLE out ERROR_VARIABLE err)
         message("${out}\n${err}")
         if(NOT "${out}${err}" MATCHES "10 test cases out of 10 passed")
            message(FATAL_ERROR "genuine executor selector did not report ten passed cases")
         endif()
      elseif(FORGE_OC_GATE STREQUAL "ordinary")
         forge_oc_native_suffix(suffix)
         execute_process(COMMAND "${FORGE_OC_CONSUMER}"
            "${directory}/add-${suffix}.request" "${directory}/add-${suffix}.foc"
            RESULT_VARIABLE result TIMEOUT 15)
      elseif(FORGE_OC_GATE STREQUAL "session" OR FORGE_OC_GATE STREQUAL "resource")
         forge_oc_absolute(FORGE_OC_HELPER)
         if(NOT FORGE_OC_AS_BYTES MATCHES "^[0-9]+$" OR FORGE_OC_AS_BYTES STREQUAL "0")
            message(FATAL_ERROR "explicit finite AS bytes required; no platform/default guess")
         endif()
         set(arguments --catch_system_errors=no)
         if(FORGE_OC_GATE STREQUAL "resource")
            list(APPEND arguments
               --run_test=optimizing_session/real_helper_explicit_absolute_as_budget_reports_outcome_after_owned_cleanup)
         endif()
         execute_process(COMMAND "${CMAKE_COMMAND}" -E env
            "FORGE_OC_COMPILER_HELPER=${FORGE_OC_HELPER}"
            "FORGE_OC_SESSION_REQUEST_DIRECTORY=${directory}"
            "FORGE_OC_HELPER_TEST_AS_BYTES=${FORGE_OC_AS_BYTES}"
            "FORGE_OC_HELPER_TEST_EXPECT_SUCCESS=1"
            "${FORGE_OC_CONSUMER}" ${arguments} RESULT_VARIABLE result TIMEOUT 120)
      else()
         message(FATAL_ERROR "unknown fixture consumer gate")
      endif()
      if(NOT result STREQUAL "0")
         message(FATAL_ERROR "fixture consumer failed: ${result}")
      endif()
   else()
      message(FATAL_ERROR "unsupported fixture operation")
   endif()
endif()
